Selected fields (with data type) are the following:

TABLE deliveries_info
    delivery_id: integer [Primary key]
    user_id: integer [Identifier of Thistle customer]
    delivery_date: date [Date of delivery. Falls on either a Monday or Thursday]
    delivery_week: date [Week of delivery. Always a Monday]
    route_id: integer [Identifier for the delivery route on which delivery occurred]
    delivery_success_flag: boolean [Marks if delivery was delivered successfully, with TRUE meaning yes]

select

case
	when date_trunc('day', delivery_date) = 1 then 'Monday'
  when date_trunc('day', delivery_date) = 4 then 'Thursday'
end as cycle_day
from deliveries_info
)
, total_delivery_cycles as (
select
delivery_week,
cycle_day,
count(delivery_id) over (partition by delivery_week, cycle_day) as total_cycle_deliveries
from deliveries_info_cycle
